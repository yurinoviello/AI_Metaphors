def normalize_text(text: str) -> str:
    text = re.sub(r'<[^>]+>\s*', '', text)
    text = re.sub(r'\s+', '', text)
    return text.strip()

class PreGeneratedSpeechService(SpeechService):
    """A SpeechService that retrieves pre-generated audio files instead of generating them."""

    def generate_from_text(self, text: str, cache_dir: str = None, path: str = None) -> dict:
        if cache_dir is None:
            cache_dir = self.cache_dir

        with open("{working_dir}/avatar/{term_name}/phrases.json", "r", encoding="utf-8") as f:
            data = json.load(f)

        text_index = None
        for key, value in data.items():
            # This checks if the script text exists inside the JSON key
            if normalize_text(text) in normalize_text(key):
                text_index = value
                break

        if text_index is None:
            raise ValueError("Text not found in phrases.json - Original: '%s', Normalized: '%s', Available keys: %s" % (text, normalize_text(text), list(data.keys())))

        audio_path = Path(f"{working_dir}/avatar/{term_name}/narration_audio/{{text_index}}.mp3")
        if not audio_path.exists():
            raise FileNotFoundError(f"Audio file not found: {{audio_path}}")

        media_path = Path(cache_dir) / audio_path
        media_path.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy(audio_path, media_path)

        return {{
            "input_text": text,
            "original_audio": str(audio_path),
        }}

    def _wrap_generate_from_text(self, text, **kwargs):
        dict_ = self.generate_from_text(text, **kwargs)
        dict_["final_audio"] = dict_["original_audio"]
        return dict_


class GenScene(VoiceoverScene):
    def remove_everything(self):
        """
        :param self: A Scene object containing all the mobjects to be removed.
        :return: None
        """
        if len(self.mobjects) > 0:
            self.play(*[FadeOut(mob) for mob in self.mobjects])
        return


    def start_end_scene(self, title: str):
        """
        :param self: The Scene object where the animations and visual elements are rendered.
        :type self: Scene
        :param title: The title text to be displayed at the start/end of the scene.
        :type title: str
        :return: None. Performs visual operations in the given scene and then exits.
        :rtype: None
        """
        # Clean the scene
        self.remove_everything()

        # Define the title and the logo
        title = Text(
            title,
            color=WHITE,
            font_size=40,
        )
        logo = ImageMobject("ai_metaphors/resources/JetBrains-logo.png").scale(0.3).next_to(title, DOWN)

        # Display the title and the logo
        self.play(Write(title))
        self.play(FadeIn(logo))
        # Leave them on the screen for 2 seconds
        self.wait(2)

        # Remove everything from the scene
        self.remove_everything()

        return


    def play_scene_with_voice(self, text: str, *animations):
        """
        Plays the given animations while narrating the provided text and displaying an animated avatar.

        Parameters
        ----------
        self : GenScene
            The current Manim scene in which animations are rendered.
        text : str
            The narration text to be spoken using a voiceover and accompanied by avatar animation.
        *animations
            A sequence of Manim animations to play in sync with the narration.

        """
        with open("{working_dir}/avatar/{term_name}/phrases.json", "r", encoding="utf-8") as f:
            data = json.load(f)

        text_index = None
        for key, value in data.items():
            if normalize_text(text) in normalize_text(key):
                text_index = value
                break

        # Exit if no matching avatar is found for the text
        if text_index is None:
            print(f"No avatar found for text: {text}")
            return

        frames_folder = f"{working_dir}/avatar/{term_name}/avatar_video_{{text_index}}/avatar_frames"
        frame_files = sorted([
            os.path.join(frames_folder, f)
            for f in os.listdir(frames_folder)
            if f.endswith(".png")
        ])
        frame_images = [ImageMobject(f).scale(0.6).to_corner(DL, 0) for f in frame_files]
        frames_count = len(frame_images)
        avatar = frame_images[0]
        self.add(avatar)
        frame_tracker = ValueTracker(0)

        def update_avatar(mob):
            idx = int(frame_tracker.get_value()) % frames_count
            mob.become(frame_images[idx])

        avatar.add_updater(update_avatar)

        self.set_speech_service(PreGeneratedSpeechService())
        with self.voiceover(text=text) as vt:
            self.play(
                frame_tracker.animate.increment_value(frames_count),
                Succession(*animations),
                run_time=vt.duration,
                rate_func=linear
            )

        avatar.remove_updater(update_avatar)
        self.play(FadeOut(avatar))
        return

    def build_scene_description(self, text: str) -> list[Animation]:
        """
        :param self: The Scene object where animations and transitions will be played.
        :param text: A string representing the description text to display on the scene.
        :return: None
        """
        description = Text(
            text,
            color=WHITE,
            font_size=24,
        ).to_edge(UP)
        return [
            Write(description),
            Wait(2),
            FadeOut(description)
        ]
