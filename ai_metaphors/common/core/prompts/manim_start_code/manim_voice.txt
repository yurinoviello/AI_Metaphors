"""This file already contains helpers that keep the narrator speaking for the entire duration of each scene.

   Each function receives the ``scene`` object (``self``) and must call Manim ``play``/``wait``/``add`` commands that build that part of the animation.
2. Call ``self.narrate(<voice‑over text>, <builder>)`` for each scene in the desired order.
3. Use ``self.describe_scene(caption)`` for short on‑screen explanations.

Intro and outro titles are handled automatically by ``start_end_scene``.

class GenScene(VoiceoverScene):
    def remove_everything(self):
        """
        :param self: A Scene object containing all the mobjects to be removed.
        :return: None
        """
        if len(self.mobjects) > 0:
            self.play(*[FadeOut(mob) for mob in self.mobjects])
        return


    def start_end_scene(self, title_text: str):
        """
        :param self: The Scene object where the animations and visual elements are rendered.
        :type self: Scene
        :param title_text: The title text to be displayed at the start/end of the scene.
        :type title_text: str
        :return: None. Performs visual operations in the given scene and then exits.
        :rtype: None
        """
        # Clean the scene
        self.remove_everything()

        # Define the title and the logo
        title_wrapped = textwrap.wrap(title_text, width=45)
        title = Paragraph(
            "\n".join(title_wrapped),
            color=WHITE,
            font_size=40,
            line_spacing=1,
            alignment='center',
        )
        logo = ImageMobject("ai_metaphors/resources/JetBrains-logo.png").scale(0.3).next_to(title, DOWN)

        self.set_speech_service(OpenAIService())
        with self.voiceover(text=title_text) as _:
            # Display the title and the logo
            self.play(Write(title))
            self.play(FadeIn(logo))
            # Leave them on the screen for 2 seconds
            self.wait(2)

        # Remove everything from the scene
        self.remove_everything()

        return

    def describe_scene(self, caption: str):
        """
        :param self: The Scene object where animations and transitions will be played.
        :param caption: A string representing the description text to display on the scene.
        :return: None
        """
        description = Text(
            caption,
            color=WHITE,
            font_size=24,
        ).to_edge(UP)

        self.play(Write(description))
        self.wait(2)
        self.play(FadeOut(description))
        return

    def narrate(self, narration_text: str, builder):
        """Speak *narration_text* while *builder* builds the visuals.
        Parameters
        ----------
        self : GenScene
            The current Manim scene in which animations are rendered.
        narration_text : str
            The exact text the narrator will read **during** the animations defined in *builder*. Extract it from the description as it is.
        builder : Callable[["GenScene"], None]
            A function that receives the scene as its only argument and issues Manim commands (``play``, ``wait``, etc.) to build the part of the animation that corresponds to *narration_text*.
            The *voice‑over* context is kept open for the whole duration of this function, so the narration spans all its animations.

        Returns
        -------
        Any
            Whatever *builder* returned (often a mobject or a tuple/dict of
            mobjects).

        Example
        -------
        >>> mailboxes = self.narrate("voice‑over", build_map)
        >>> # *mailboxes* can now be reused in later scenes
        """
        with self.voiceover(text=narration_text):
            result = builder(self)
        self.wait(2)
        return result
